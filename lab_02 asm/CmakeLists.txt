cmake_minimum_required(VERSION 3.10)
project(Lab2)

set(CMAKE_CXX_STANDARD 17)

find_package(spdlog REQUIRED)

add_executable(lab2 main.cpp lab2_lib.cpp)
target_link_libraries(lab2 PRIVATE spdlog::spdlog)

find_package(GTest REQUIRED)
add_executable(tests tests.cpp lab2_lib.cpp)
target_link_libraries(tests PRIVATE GTest::GTest GTest::Main spdlog::spdlog)
include(GoogleTest)
gtest_discover_tests(tests)
